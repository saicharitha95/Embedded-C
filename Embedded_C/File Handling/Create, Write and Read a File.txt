#include <stdio.h>

int main()
{
    FILE *fp;
    char filename[100];
    char text[500];

    printf("Enter file name: ");
    scanf("%99s", filename);

    getchar();

    printf("Enter text: ");
    fgets(text, sizeof(text), stdin);

    fp = fopen(filename, "w");

    if (fp == NULL)
    {
        printf("Error: Unable to create file.\n");
        return 1;
    }

    fputs(text, fp);
    fclose(fp);

    printf("File '%s' created and saved.\n", filename);

    fp = fopen(filename, "r");

    if (fp == NULL)
    {
        printf("Error: Unable to open file.\n");
        return 1;
    }

    printf("\nFile contents:\n");

    while (fgets(text, sizeof(text), fp) != NULL)
    {
        printf("%s", text);
    }

    fclose(fp);

    return 0;
}
