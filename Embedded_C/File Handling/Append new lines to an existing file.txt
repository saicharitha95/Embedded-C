#include <stdio.h>

int main()
{
    FILE *fp;
    char filename[100];
    char text[500];
    int n, i;

    printf("Enter file name: ");
    scanf("%99s", filename);

    fp = fopen(filename, "a");

    if (fp == NULL)
    {
        printf("Error: Unable to open file.\n");
        return 1;
    }

    printf("Enter number of lines to add: ");
    scanf("%d", &n);

    getchar();

    for (i = 0; i < n; i++)
    {
        printf("Enter line %d: ", i + 1);
        fgets(text, sizeof(text), stdin);
        fputs(text, fp);
    }

    fclose(fp);

    printf("New lines added to '%s'.\n", filename);

    return 0;
}
