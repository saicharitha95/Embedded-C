#include <stdio.h>

void display(int *arr, int n)
{
    int *p = arr;

    for (int i = 0; i < n; i++)
        printf("%d ", *(p + i));

    printf("\n");
}

int main(void)
{
    int n;
    int arr[100];

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (int i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Array elements:\n");
    display(arr, n);

    return 0;
}
