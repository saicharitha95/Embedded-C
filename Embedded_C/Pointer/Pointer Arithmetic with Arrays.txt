#include <stdio.h>

int main(void)
{
    int n;
    int arr[100];
    int *p;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (int i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    p = arr;

    printf("Array elements using pointer arithmetic:\n");

    for (int i = 0; i < n; i++)
        printf("%d ", *(p + i));

    printf("\n");

    return 0;
}
