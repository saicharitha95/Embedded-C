#include <stdio.h>

int isEven(int num);

int main(void)
{
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);

    if (isEven(num))
    {
        printf("%d is Even\n", num);
    }
    else
    {
        printf("%d is Odd\n", num);
    }

    return 0;
}

int isEven(int num)
{
    if (num % 2 == 0)
        return 1;
    else
        return 0;
}
