#include <stdio.h>

int isPrime(int num);

int main(void)
{
    int num;

    printf("Enter a number: ");
    scanf("%d", &num);

    if (isPrime(num))
    {
        printf("%d is a Prime number\n", num);
    }
    else
    {
        printf("%d is Not a Prime number\n", num);
    }

    return 0;
}

int isPrime(int num)
{
    int i;

    if (num <= 1)
    {
        return 0;
    }

    for (i = 2; i <= num / 2; i++)
    {
        if (num % i == 0)
        {
            return 0;
        }
    }

    return 1;
}
