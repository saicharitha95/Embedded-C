#include <stdio.h>

long long factorial(int n);
long long power(int base, int exponent);

int main(void)
{
    int num;
    int base, exponent;

    printf("Enter a number for factorial: ");
    scanf("%d", &num);

    if (num < 0)
    {
        printf("Factorial is not possible for negative numbers\n");
    }
    else
    {
        printf("Factorial of %d = %lld\n", num, factorial(num));
    }

    printf("\nEnter base: ");
    scanf("%d", &base);

    printf("Enter exponent: ");
    scanf("%d", &exponent);

    if (exponent < 0)
    {
        printf("Negative exponent is not supported\n");
    }
    else
    {
        printf("%d^%d = %lld\n",
               base, exponent, power(base, exponent));
    }

    return 0;
}

long long factorial(int n)
{
    if (n == 0 || n == 1)
    {
        return 1;
    }

    return n * factorial(n - 1);
}

long long power(int base, int exponent)
{
    if (exponent == 0)
    {
        return 1;
    }

    return base * power(base, exponent - 1);
}
