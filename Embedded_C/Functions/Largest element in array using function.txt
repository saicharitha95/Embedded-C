#include <stdio.h>

int findLargest(int arr[], int size);

int main(void)
{
    int arr[100];
    int n, i, largest;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements:\n", n);

    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    largest = findLargest(arr, n);

    printf("Largest element = %d\n", largest);

    return 0;
}

int findLargest(int arr[], int size)
{
    int i;
    int largest = arr[0];

    for (i = 1; i < size; i++)
    {
        if (arr[i] > largest)
        {
            largest = arr[i];
        }
    }

    return largest;
}
