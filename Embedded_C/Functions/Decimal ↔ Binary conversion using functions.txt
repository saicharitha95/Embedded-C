#include <stdio.h>

void decimalToBinary(int decimal);
int binaryToDecimal(long long binary);

int main(void)
{
    int choice;
    int decimal;
    long long binary;
    int result;

    printf("1. Decimal to Binary\n");
    printf("2. Binary to Decimal\n");

    printf("Enter your choice: ");
    scanf("%d", &choice);

    if (choice == 1)
    {
        printf("Enter decimal number: ");
        scanf("%d", &decimal);

        printf("Binary = ");
        decimalToBinary(decimal);
        printf("\n");
    }
    else if (choice == 2)
    {
        printf("Enter binary number: ");
        scanf("%lld", &binary);

        result = binaryToDecimal(binary);

        if (result == -1)
        {
            printf("Invalid binary number\n");
        }
        else
        {
            printf("Decimal = %d\n", result);
        }
    }
    else
    {
        printf("Invalid choice\n");
    }

    return 0;
}

void decimalToBinary(int decimal)
{
    int binary[32];
    int i = 0;

    if (decimal == 0)
    {
        printf("0");
        return;
    }

    while (decimal > 0)
    {
        binary[i] = decimal % 2;
        decimal = decimal / 2;
        i++;
    }

    while (i > 0)
    {
        i--;
        printf("%d", binary[i]);
    }
}

int binaryToDecimal(long long binary)
{
    int decimal = 0;
    int base = 1;
    int digit;

    if (binary == 0)
    {
        return 0;
    }

    while (binary > 0)
    {
        digit = binary % 10;

        if (digit != 0 && digit != 1)
        {
            return -1;
        }

        decimal = decimal + digit * base;
        base = base * 2;

        binary = binary / 10;
    }

    return decimal;
}
