#include <stdio.h>

void increment(int *value)
{
    (*value)++;
}

void doubleValue(int *value)
{
    (*value) *= 2;
}

void processArray(int arr[], int n, void (*callback)(int *))
{
    for (int i = 0; i < n; i++)
        callback(&arr[i]);
}

int main(void)
{
    int n, choice;
    int arr[100];

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");

    for (int i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("\n1. Increment each element\n");
    printf("2. Double each element\n");

    printf("Enter your choice: ");
    scanf("%d", &choice);

    if (choice == 1)
    {
        processArray(arr, n, increment);
    }
    else if (choice == 2)
    {
        processArray(arr, n, doubleValue);
    }
    else
    {
        printf("Invalid choice\n");
        return 0;
    }

    printf("Modified array:\n");

    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);

    printf("\n");

    return 0;
}
