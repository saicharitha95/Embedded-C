#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    int n, newSize;
    int *arr;
    int *temp;

    printf("Enter initial number of elements: ");
    scanf("%d", &n);

    arr = malloc(n * sizeof(int));

    if (arr == NULL)
    {
        printf("Memory allocation failed\n");
        return 1;
    }

    printf("Enter %d elements:\n", n);

    for (int i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }

    printf("Enter new size: ");
    scanf("%d", &newSize);

    if (newSize <= n)
    {
        printf("New size must be greater than %d\n", n);
        free(arr);
        return 0;
    }

    temp = realloc(arr, newSize * sizeof(int));

    if (temp == NULL)
    {
        printf("Memory reallocation failed\n");
        free(arr);
        return 1;
    }

    arr = temp;

    printf("Enter %d additional elements:\n", newSize - n);

    for (int i = n; i < newSize; i++)
    {
        scanf("%d", &arr[i]);
    }

    printf("Array after resizing:\n");

    for (int i = 0; i < newSize; i++)
    {
        printf("%d ", arr[i]);
    }

    printf("\n");

    free(arr);
    arr = NULL;

    return 0;
}
