#include <stdio.h>
#include <stdint.h>

#define SET_BIT(REG, BIT)       ((REG) |= (1U << (BIT)))
#define CLEAR_BIT(REG, BIT)     ((REG) &= ~(1U << (BIT)))
#define TOGGLE_BIT(REG, BIT)    ((REG) ^= (1U << (BIT)))
#define CHECK_BIT(REG, BIT)     (((REG) >> (BIT)) & 1U)

int main(void)
{
    uint32_t num;
    unsigned int n;

    printf("Enter a number: ");
    scanf("%u", &num);

    printf("Enter bit position (0-31): ");
    scanf("%u", &n);

    if (n > 31)
    {
        printf("Invalid bit position\n");
        return 0;
    }

    printf("Original value = %u\n", num);

    SET_BIT(num, n);
    printf("After SET = %u\n", num);

    CLEAR_BIT(num, n);
    printf("After CLEAR = %u\n", num);

    TOGGLE_BIT(num, n);
    printf("After TOGGLE = %u\n", num);

    printf("Bit %u status = %u\n", n, CHECK_BIT(num, n));

    return 0;
}
