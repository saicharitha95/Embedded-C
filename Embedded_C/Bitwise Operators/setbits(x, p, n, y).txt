#include <stdio.h>
#include <stdint.h>

uint32_t setbits(uint32_t x, int p, int n, uint32_t y)
{
    uint32_t mask;
    uint32_t ybits;

    if (n == 32)
        mask = 0xFFFFFFFFU;
    else
        mask = (1U << n) - 1U;

    ybits = y & mask;
    ybits = ybits << (p - n + 1);

    mask = mask << (p - n + 1);
    x = x & ~mask;

    x = x | ybits;

    return x;
}

int main(void)
{
    uint32_t x, y, result;
    int p, n;

    printf("Enter x: ");
    scanf("%u", &x);

    printf("Enter y: ");
    scanf("%u", &y);

    printf("Enter position p: ");
    scanf("%d", &p);

    printf("Enter number of bits n: ");
    scanf("%d", &n);

    if (p < 0 || p > 31 || n <= 0 || n > 32 || n > p + 1)
    {
        printf("Invalid values\n");
        return 0;
    }

    result = setbits(x, p, n, y);

    printf("Result = %u\n", result);

    return 0;
}
