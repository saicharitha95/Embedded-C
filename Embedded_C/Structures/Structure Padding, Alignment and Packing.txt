#include <stdio.h>
#include <stddef.h>

struct Normal
{
    char c;
    int i;
    short s;
};

#pragma pack(push, 1)

struct Packed
{
    char c;
    int i;
    short s;
};

#pragma pack(pop)

int main(void)
{
    struct Normal normal;
    struct Packed packed;

    printf("Normal Structure\n");
    printf("----------------\n");
    printf("Size = %zu bytes\n", sizeof(normal));
    printf("Offset of c = %zu\n", offsetof(struct Normal, c));
    printf("Offset of i = %zu\n", offsetof(struct Normal, i));
    printf("Offset of s = %zu\n", offsetof(struct Normal, s));

    printf("\nPacked Structure\n");
    printf("----------------\n");
    printf("Size = %zu bytes\n", sizeof(packed));
    printf("Offset of c = %zu\n", offsetof(struct Packed, c));
    printf("Offset of i = %zu\n", offsetof(struct Packed, i));
    printf("Offset of s = %zu\n", offsetof(struct Packed, s));

    return 0;
}
