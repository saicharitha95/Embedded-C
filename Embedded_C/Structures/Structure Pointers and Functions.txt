#include <stdio.h>

struct Sensor
{
    int id;
    float temperature;
    int status;
};

void displaySensor(struct Sensor *sensor)
{
    printf("\nSensor Information\n");
    printf("ID = %d\n", sensor->id);
    printf("Temperature = %.2f\n", sensor->temperature);
    printf("Status = %d\n", sensor->status);
}

void updateSensor(struct Sensor *sensor)
{
    printf("\nEnter new temperature: ");
    scanf("%f", &sensor->temperature);

    printf("Enter new status: ");
    scanf("%d", &sensor->status);
}

int main(void)
{
    struct Sensor sensor;
    struct Sensor *ptr = &sensor;

    printf("Enter sensor ID: ");
    scanf("%d", &ptr->id);

    printf("Enter temperature: ");
    scanf("%f", &ptr->temperature);

    printf("Enter status: ");
    scanf("%d", &ptr->status);

    displaySensor(ptr);

    updateSensor(ptr);

    printf("\nUpdated Sensor Information\n");
    displaySensor(ptr);

    return 0;
}
