#include <stdio.h>

struct Data
{
    char c;
    int i;
    float f;
};

union DataUnion
{
    char c;
    int i;
    float f;
};

int main(void)
{
    struct Data s;
    union DataUnion u;

    printf("Size of structure = %zu bytes\n", sizeof(s));
    printf("Size of union = %zu bytes\n", sizeof(u));

    printf("\nEnter character: ");
    scanf(" %c", &s.c);

    printf("Enter integer: ");
    scanf("%d", &s.i);

    printf("Enter float: ");
    scanf("%f", &s.f);

    printf("\nStructure values:\n");
    printf("Character = %c\n", s.c);
    printf("Integer = %d\n", s.i);
    printf("Float = %.2f\n", s.f);

    printf("\nEnter character for union: ");
    scanf(" %c", &u.c);
    printf("Union character = %c\n", u.c);

    printf("Enter integer for union: ");
    scanf("%d", &u.i);
    printf("Union integer = %d\n", u.i);

    printf("Enter float for union: ");
    scanf("%f", &u.f);
    printf("Union float = %.2f\n", u.f);

    printf("\nSize of structure = %zu bytes\n", sizeof(s));
    printf("Size of union = %zu bytes\n", sizeof(u));

    return 0;
}
