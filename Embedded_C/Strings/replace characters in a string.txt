#include <stdio.h>

int main()
{
    char str[100];
    char old_char, new_char;
    int i;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    printf("Enter the character to replace: ");
    scanf(" %c", &old_char);

    printf("Enter the new character: ");
    scanf(" %c", &new_char);

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] == old_char)
        {
            str[i] = new_char;
        }
    }

    printf("String after replacement: %s\n", str);

    return 0;
}
