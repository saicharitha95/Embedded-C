#include <stdio.h>

int main()
{
    char str[200];
    int i;
    int words = 0;
    int in_word = 0;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] != ' ' && in_word == 0)
        {
            words++;
            in_word = 1;
        }
        else if (str[i] == ' ')
        {
            in_word = 0;
        }
    }

    printf("Total number of words = %d\n", words);

    return 0;
}
