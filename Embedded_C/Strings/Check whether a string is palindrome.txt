#include <stdio.h>

int main()
{
    char str[100];
    int length = 0;
    int i;
    int palindrome = 1;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    /* Find length */
    while (str[length] != '\0')
    {
        length++;
    }

    /* Compare characters from both ends */
    for (i = 0; i < length / 2; i++)
    {
        if (str[i] != str[length - 1 - i])
        {
            palindrome = 0;
            break;
        }
    }

    if (palindrome == 1)
    {
        printf("String is a Palindrome\n");
    }
    else
    {
        printf("String is not a Palindrome\n");
    }

    return 0;
}
