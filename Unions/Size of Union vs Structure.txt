#include <stdio.h>

union Data
{
    int i;
    double d;
    char c;
};

struct DataStructure
{
    int i;
    double d;
    char c;
};

int main(void)
{
    union Data u;
    struct DataStructure s;

    printf("Size of int = %zu bytes\n", sizeof(int));
    printf("Size of double = %zu bytes\n", sizeof(double));
    printf("Size of char = %zu bytes\n", sizeof(char));

    printf("\nSize of union = %zu bytes\n", sizeof(u));
    printf("Size of structure = %zu bytes\n", sizeof(s));

    printf("\nEnter integer: ");
    scanf("%d", &u.i);
    printf("Union integer = %d\n", u.i);

    printf("Enter double: ");
    scanf("%lf", &u.d);
    printf("Union double = %.2lf\n", u.d);

    printf("Enter character: ");
    scanf(" %c", &u.c);
    printf("Union character = %c\n", u.c);

    return 0;
}
